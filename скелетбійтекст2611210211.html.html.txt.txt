const V = {
  sub:  (a,b)=>[a[0]-b[0],a[1]-b[1],a[2]-b[2]],
  add:  (a,b)=>[a[0]+b[0],a[1]+b[1],a[2]+b[2]],
  cross:(a,b)=>[a[1]*b[2]-a[2]*b[1],a[2]*b[0]-a[0]*b[2],a[0]*b[1]-a[1]*b[0]],
  norm: a=>{const l=Math.hypot(a[0],a[1],a[2])||1;return[a[0]/l,a[1]/l,a[2]/l];}
};
const IDENT = [1,0,0, 0,1,0, 0,0,1];
function mMul(a,b){
  const r=new Array(9);
  for(let i=0;i<3;i++)for(let j=0;j<3;j++)
    r[i*3+j]=a[i*3]*b[j]+a[i*3+1]*b[3+j]+a[i*3+2]*b[6+j];
  return r;
}
function mApply(m,v){
  return [m[0]*v[0]+m[1]*v[1]+m[2]*v[2],
          m[3]*v[0]+m[4]*v[1]+m[5]*v[2],
          m[6]*v[0]+m[7]*v[1]+m[8]*v[2]];
}
function mRot(rx,ry,rz){
  const cx=Math.cos(rx),sx=Math.sin(rx);
  const cy=Math.cos(ry),sy=Math.sin(ry);
  const cz=Math.cos(rz),sz=Math.sin(rz);
  return mMul(mMul([cy,0,sy,0,1,0,-sy,0,cy],[1,0,0,0,cx,-sx,0,sx,cx]),
              [cz,-sz,0,sz,cz,0,0,0,1]);
}

const cam = { pos:[0,2.25,6.1], target:[0,1.15,0], up:[0,1,0] };
let basis = { f:[0,0,-1], r:[1,0,0], u:[0,1,0] };
function updateBasis(){
  const f = V.norm(V.sub(cam.target, cam.pos));
  const r = V.norm(V.cross(f, cam.up));
  basis.f = f; basis.r = r; basis.u = V.cross(r, f);
}
function project(p){
  const dx=p[0]-cam.pos[0],dy=p[1]-cam.pos[1],dz=p[2]-cam.pos[2];
  const z = dx*basis.f[0]+dy*basis.f[1]+dz*basis.f[2];
  if (z < 0.12) return null;
  const s = FOCAL/z;
  return { x: CX+(dx*basis.r[0]+dy*basis.r[1]+dz*basis.r[2])*s,
           y: CY-(dx*basis.u[0]+dy*basis.u[1]+dz*basis.u[2])*s, s, z };
}

function mkNode(name, off, parent){
  const n={ name, off, rx:0, ry:0, rz:0, children:[], pos:[0,0,0], mat:IDENT.slice() };

}

function updateNode(n, parentPos, parentMat){
  const local = mRot(n.rx, n.ry, n.rz);
  const off = mApply(parentMat, n.off);
  n.pos[0]=parentPos[0]+off[0];
  n.pos[1]=parentPos[1]+off[1];
  n.pos[2]=parentPos[2]+off[2];
  n.mat = mMul(parentMat, local);
  for (let i=0;i<n.children.length;i++) updateNode(n.children[i], n.pos, n.mat);
}
const HIP_Y = 0.95;
function updateSkeleton(f){
  const j=f.joints;
  const fall = (f.state==='dead') ? f.deathK : 0;
  const fe = fall*fall;
  const yaw  = f.yaw + (f.spinYaw||0);
  const tilt = (f.tilt||0);
  const pitch = (f.pitch||0);
  const rootMat = mMul(mMul(mRot(0,yaw,0), mRot(pitch,0,tilt)), mRot(-1.5*fe,0,0));
  const baseY = HIP_Y - 0.70*fe + f.yOff;
  const rootPos = [f.x, baseY, 0];
  j.root.pos = rootPos; j.root.mat = rootMat;
  for (let i=0;i<j.root.children.length;i++) updateNode(j.root.children[i], rootPos, rootMat);
}

function pickWeighted(list){
  let total = 0;
  for (const a of list) total += a.w;
  let r = Math.random() * total;
  for (const a of list) { r -= a.w; if (r <= 0) return a; }
  return list[0];
}

        break;
      }

      /* ---- МАЙК: ПАДІННЯ НА РУКИ + НОГИ ДІАГОНАЛЬНО ---- */
      case 'hands_fall': {
        j.shL.rx = -1.35*k; j.shR.rx = -1.35*k;
        j.elL.rx = 0.1;    j.elR.rx = 0.1;
        j.spine1.rx = 0.9*k; j.chest.rx = 0.35*k;
        j.head.rx = -0.5*k;
        j.hipL.rx = -1.1*k; j.hipR.rx = -1.15*k;
        j.knL.rx  = 0.15;   j.knR.rx = 0.12;
        // тремтіння
        const tr = Math.sin(this.t*32)*0.06*k;
        j.hipL.rz = -0.35*k + tr;
        j.hipR.rz =  0.35*k - tr;
        this.yOff += -0.35*k;
        this.pitch = 0.9*k;
        break;
      }

      /* ---- ДОН: СТРИБОК ЧЕРЕЗ ПЛАНКУ + УДАР 2 НОГАМИ ЗГОРИ ---- */
      case 'hurdle_strike': {
        // фаза 1: розбіг і стрибок через планку
        if (p < 0.55) {
          const kk = p/0.55;
          j.hipL.rx = -1.3*kk; j.hipR.rx = -1.3*kk;
          j.knL.rx  = 1.2*kk;  j.knR.rx  = 1.2*kk;
          j.shL.rx  = -1.0*kk; j.shR.rx  = -1.0*kk;
          j.spine1.rx = 0.5*kk;
          j.head.rx = -0.4*kk;
          this.pitch = 0.65*kk;
        } else {
          // фаза 2: удар згори двома ногами
          const kk = (p-0.55)/0.45;
          const strike = easeOut(Math.min(1,kk*1.6));
          j.hipL.rx = -0.5*strike; j.hipR.rx = -0.55*strike;
          j.knL.rx  = 1.3*(1-strike)+0.1;
          j.knR.rx  = 1.3*(1-strike)+0.1;
          j.shL.rx = 0.7*strike; j.shR.rx = 0.7*strike;
          j.spine1.rx = 0.35*strike;
          this.pitch = 0.65*(1-strike);
        }
        break;
      }

      /* ---- КЕЙСІ: СТРИБОК ГОЛОВОЮ ВНИЗ + УДАР ---- */
      case 'head_down_jump': {
        if (p < 0.35) {
          const kk = p/0.35;
          j.hipL.rx = -1.4*kk; j.hipR.rx = -1.3*kk;
          j.knL.rx  = 1.3*kk;  j.knR.rx  = 1.4*kk;
          j.hipL.rz = 0.3*kk*Math.sin(this.t*18);
          j.hipR.rz = -0.3*kk*Math.sin(this.t*18);
          this.pitch = 1.5*kk;
          j.head.rx = 0.4*kk;
          j.shL.rx = -0.8*kk; j.shR.rx = -0.8*kk;
        } else {
          const kk = (p-0.35)/0.65;
          const strike = easeOut(Math.min(1,kk*2.2));
          this.pitch = 1.5*(1-strike);
          const side=this.armSide;
          const sh = side===1?j.shR:j.shL;
          sh.rx = -1.7*strike;
          j.shL.rx += -0.5*strike;
          j.shR.rx += -0.5*strike;
          j.spine1.rx = 0.4*strike;
          j.hipL.rx = -0.3*strike;
          j.hipR.rx = -0.3*strike;
          j.knL.rx  = 0.2; j.knR.rx = 0.2;
        }
        break;
      }

      /* ---- ДРАКОН: НАХИЛ НАЗАД + УДАР ГОЛОВОЮ ---- */
      case 'dragon_headbutt': {
        if (p < 0.35) {
          const kk = p/0.35;
          j.spine1.rx = 0.55*kk;
          j.spine2.rx = 0.35*kk;
          j.chest.rx  = 0.25*kk;
          j.head.rx   = 0.5*kk;
          j.shL.rx    = -0.5*kk; j.shR.rx = -0.5*kk;
          this.pitch = 0.25*kk;
        } else {
          const kk = (p-0.35)/0.65;
          const strike = easeOut(Math.min(1,kk*2.4));
          j.spine1.rx = 0.55 - 1.2*strike;
          j.spine2.rx = 0.35 - 0.6*strike;
          j.chest.rx  = 0.25 - 0.4*strike;
          j.head.rx   = 0.5 - 1.2*strike;
          j.shL.rx = 0.4*strike; j.shR.rx = 0.4*strike;
          j.hipL.rx = -0.2*strike; j.hipR.rx = -0.2*strike;
          this.pitch = 0.25 - 0.4*strike;
        }
        break;
      }

      /* ---- ШРЕДЕР: РУКИ ПО ЛИЦЮ ---- */
      case 'shred_face': {
        j.shL.rx = -1.35*k; j.shR.rx = -1.35*k;
        j.shL.rz = -0.14 - 0.4*k; j.shR.rz = 0.14 + 0.4*k;
        j.elL.rx = 0.1; j.elR.rx = 0.1;
        // "рве" – коливання
        const tr = Math.sin(this.t*36)*0.12*k;
        j.haL.rx = tr; j.haR.rx = -tr;
        j.spine1.rx = 0.25*k; j.head.rx = 0.2*k;
        j.hipL.rx = -0.15*k; j.hipR.rx = -0.15*k;
        this.yOff += -0.03*k;
        break;
      }

      /* ---- MK: ПІДСІЧКА ПІД НОГИ ---- */
      case 'sweep': {
        const side=this.legSide;
        const hip=side===1?j.hipR:j.hipL;
        const kn =side===1?j.knR :j.knL;
        hip.rx = -1.4*k;
        hip.rz = side*0.55*k;
        kn.rx  = 0.1;
        j.spine1.rx = 0.35*k;
        j.spine1.rz = -side*0.3*k;
        j.shL.rx = 0.5*k; j.shR.rx = 0.5*k;
        this.yOff += -0.28*k;
        break;
      }

      /* ---- КИДОК: ПІДНЯТИ І КИНУТИ ---- */
      case 'throw': {
        if (p < 0.4) {
          const kk = p/0.4;
          j.shL.rx = -1.1*kk; j.shR.rx = -1.1*kk;
          j.elL.rx = 0.2; j.elR.rx = 0.2;
          j.spine1.rx = 0.5*kk;
          j.hipL.rx = -0.3*kk; j.hipR.rx = -0.3*kk;
          j.knL.rx  = 0.5*kk;  j.knR.rx  = 0.5*kk;
          this.yOff += -0.15*kk;
        } else if (p < 0.7) {
          const kk = (p-0.4)/0.3;
          j.shL.rx = -1.9*kk - 0.4; j.shR.rx = -1.9*kk - 0.4;
          j.spine1.rx = 0.5 - 0.9*kk;
          this.yOff += -0.15 + 0.2*kk;
        } else {
          const kk = (p-0.7)/0.3;
          const strike = easeOut(kk);
          j.shL.rx = -1.2*(1-strike); j.shR.rx = -1.2*(1-strike);
          j.spine1.rx = -0.4 + 0.4*strike;
          this.yOff += 0.05*(1-strike);
        }
        break;
      }

      /* ---- СТРИБОК: 2 НОГИ ВПЕРЕД, ТІЛО ВГОРУ, КУТ 0-55° ---- */
      case 'jump_kick_2legs': {
        const tiltMax = 0.96; // ~55°
        const tiltAng = tiltMax * (0.3 + this.variant*0.7);
        j.hipL.rx = -1.2*k; j.hipR.rx = -1.2*k;
        j.knL.rx  = 0.15; j.knR.rx  = 0.15;
        j.hipL.rz = -0.15*k; j.hipR.rz = 0.15*k;
        j.shL.rx = 0.5*k; j.shR.rx = 0.5*k;
        j.shL.rz = -0.4*k; j.shR.rz = 0.4*k;
        j.spine1.rx = -tiltAng*k;
        j.head.rx   = tiltAng*0.6*k;
        this.pitch = -tiltAng*k;
        break;
      }

      /* ---- ЄВРЕЙСЬКИЙ УДАР В ОЧІ ---- */
      case 'eye_strike': {
        j.shL.rx = -1.62*k; j.shR.rx = -1.62*k;
        j.elL.rx = 0.05; j.elR.rx = 0.05;
        j.shL.rz = -0.05; j.shR.rz = 0.05;
        j.spine1.rx = 0.15*k;
        j.head.rx = 0.1*k;
        break;
      }

      /* ---- 2 НОГИ В КОЛІНО ---- */
      case 'double_knee': {
        j.hipL.rx = -1.3*k; j.hipR.rx = -1.3*k;
        j.knL.rx  = 1.35*k; j.knR.rx  = 1.35*k;
        j.shL.rx = 0.35*k; j.shR.rx = 0.35*k;
        j.spine1.rx = 0.2*k;
        j.head.rx = -0.15*k;
        this.yOff += 0.1*k;
        break;
      }

      /* ---- ВИРВАТИ ЗУБ ---- */
      case 'rip_tooth': {
        const side=this.armSide;
        const sh = side===1?j.shR:j.shL;
        const el = side===1?j.elR:j.elL;
        sh.rx = -1.5*k;
        sh.rz = side*(0.14 - 0.25*k);
        el.rx = 0.15*k;
        // ривок назад
        const rip = Math.sin(clamp(this.t/dur,0,1)*Math.PI);
        j.spine1.rx = 0.15*k - 0.25*rip;
        j.head.rx = 0.2*rip;
        j.hipL.rx = -0.1*k; j.hipR.rx = -0.1*k;
        break;
      }
    }
  }

  /* ---- логіка ---- */
  startAttack(){
    const atk = pickWeighted(ATTACKS);
    this.atk = atk;
    this.hitDone = false;
    this.armSide = Math.random()<0.5 ? 1 : -1;
    this.legSide = Math.random()<0.5 ? 1 : -1;
    this.variant = Math.random();
    this.spinYaw = 0;
    this.setState(atk.name);
    showLog(atk.name);
  }

  takeHit(dmg, opp){
    if (this.state==='dead') return;
    this.hp -= dmg;
    hitFlash = Math.min(1, hitFlash + 0.8);

    // іскри
    const px=(this.x+opp.x)*0.5;
    const pp=project([px, 1.0 + opp.target*0.55, 0]);
    if (pp) spawnSparks(pp.x, pp.y, this.color, 10);

    if (this.hp <= 0){
      this.hp = 0;
      this.deathK = 0;
      this.setState('dead');
    } else {
      this.setState('hurt');
      this.x -= this.faceDir * 0.12;
    }
  }

  /* спеціальний ефект – поштовх назад / ковзання */
  applyKnock(knockType){
    if (knockType==='sweep'){
      this.knockT = 0.55;
      this.knockVX = -this.faceDir * 2.2;
    } else if (knockType==='throw'){
      this.knockT = 0.8;
      this.knockVX = -this.faceDir * 4.5;
    }
  }

  update(dt, opp){
    this.animT += dt; this.t += dt;

    // затухання спіну
    if (this.spinYaw && this.state!=='spin_kick'){
      this.spinYaw *= Math.pow(0.02, dt);
      if (Math.abs(this.spinYaw)<0.01) this.spinYaw=0;
    }

    // поштовх / ковзання
    if (this.knockT > 0){
      this.knockT -= dt;
      this.x += this.knockVX * dt;
      this.knockVX *= (1 - 2.5*dt);
      this.x = clamp(this.x, -5.0, 5.0);
      if (this.knockT <= 0) this.setState('idle');
      // залишаємось у стані «swept» без руху
      this.resetPose();
      this.poseHurt();
      this.yOff = -0.4 * Math.max(0,this.knockT);
      return;
    }

    const targetYaw = this.faceDir > 0 ? Math.PI/2 : -Math.PI/2;
    this.yaw += (targetYaw - this.yaw) * Math.min(1, dt*16);

    if (this.state === 'dead'){
      this.deathK = Math.min(1, this.deathK + dt*1.7);
      return;
    }

    this.resetPose();

    const dx = opp.x - this.x;
    const dist = Math.abs(dx);
    const toward = dx >= 0 ? 1 : -1;
    if (toward !== this.faceDir) this.faceDir = toward;

    switch (this.state) {

      case 'idle': {
        this.poseIdle();
        if (this.t > this.idleWait){
          if (dist > 1.35) this.setState('approach');
          else this.startAttack();
        }
        break;
      }

      case 'approach': {
        this.x += this.faceDir * 3.6 * dt;
        this.poseWalk(this.animT * 12);
        if (dist <= 1.22){
          this.setState('idle');
          this.idleWait = 0.02;
        }
        break;
      }

      case 'hurt': {
        this.poseHurt();
        if (this.t > 0.22){
          this.setState('idle');
          this.idleWait = 0.03 + Math.random()*0.16;
        }
        break;
      }

      default: {
        // атака
        const a = this.atk;
        if (!a){ this.setState('idle'); break; }

        if (!this.hitDone && this.t >= a.hit){
          this.hitDone = true;
          if (dist < a.range){
            const dmg = a.dmg[0] + Math.random()*(a.dmg[1]-a.dmg[0]);
            opp.takeHit(dmg, this);
            if (a.knock) opp.applyKnock(a.knock);
          }
        }
        this.poseAttack();
        if (this.t >= a.dur){
          this.setState('idle');
          this.idleWait = 0.04 + Math.random()*0.22;
        }
        break;
      }
    }

    this.x = clamp(this.x, -5.0, 5.0);
  }
}

/* ---- частинки ---- */
const sparks = [];
function spawnSparks(x, y, color, n){
  for (let i=0; i<n; i++){
    const a = Math.random()*Math.PI*2;
    const sp = 80 + Math.random()*320;
    sparks.push({
      x, y,
      vx: Math.cos(a)*sp,
      vy: Math.sin(a)*sp - 80,
      life: 0, max: 0.28 + Math.random()*0.32,
      color, size: 1.3 + Math.random()*2.6
    });
  }
}
function updateSparks(dt){
  for (let i=sparks.length-1; i>=0; i--){
    const p=sparks[i];
    p.life += dt;
    if (p.life >= p.max){ sparks.splice(i,1); continue; }
    p.vy += 950*dt;
    p.x += p.vx*dt; p.y += p.vy*dt;
    p.vx *= (1 - 2.2*dt);
  }
}
function drawSparks(){
  ctx.globalCompositeOperation = 'lighter';
  for (const p of sparks){
    const k = 1 - p.life/p.max;
    ctx.globalAlpha = k;
    ctx.fillStyle = p.color;
    ctx.beginPath();
    ctx.arc(p.x, p.y, p.size*k, 0, Math.PI*2);
    ctx.fill();
  }
  ctx.globalAlpha = 1;
  ctx.globalCompositeOperation = 'source-over';
}

let shake = 0;

function drawFloor(){
  const R = 6;
  for (let ring=1; ring<=3; ring++){
    const r = R*ring/3;
    ctx.beginPath();
    let started = false;
    for (let i=0; i<=72; i++){
      const a = i/72*Math.PI*2;
      const p = project([Math.cos(a)*r, 0, Math.sin(a)*r]);
      if (!p){ started=false; continue; }
      if (!started){ ctx.moveTo(p.x,p.y); started=true; }
      else ctx.lineTo(p.x,p.y);
    }
    ctx.lineWidth = ring===3 ? 1.8 : 1;
    ctx.strokeStyle = ring===3 ? 'rgba(90,140,255,0.55)' : 'rgba(60,90,170,0.22)';
    ctx.stroke();
  }
  ctx.lineWidth = 1;
  ctx.strokeStyle = 'rgba(60,90,170,0.16)';
  const c0 = project([0,0,0]);
  if (c0){
    for (let i=0; i<12; i++){
      const a = i/12*Math.PI*2;
      const p2 = project([Math.cos(a)*R, 0, Math.sin(a)*R]);
      if (!p2) continue;
      ctx.beginPath(); ctx.moveTo(c0.x,c0.y); ctx.lineTo(p2.x,p2.y); ctx.stroke();
    }
  }
}

function drawShadow(f){
  const p = project([f.x, 0.01, 0]);
  if (!p) return;
  const rx = 0.55 * p.s;
  const ry = rx * 0.30;
  const g = ctx.createRadialGradient(p.x,p.y,0, p.x,p.y,rx);
  g.addColorStop(0,'rgba(0,0,0,0.55)');
  g.addColorStop(0.6,'rgba(0,0,0,0.22)');
  g.addColorStop(1,'rgba(0,0,0,0)');
  ctx.fillStyle = g;
  ctx.beginPath();
  ctx.ellipse(p.x,p.y,rx,ry,0,0,Math.PI*2);
  ctx.fill();
}

function addFighterItems(f, items){
  const j=f.joints;
  const col=f.color;
  const line = (a,b,w)=>items.push({ type:'line', a:a.pos, b:b.pos, color:col, w:w||1 });

  line(j.pelvis,j.spine1); line(j.spine1,j.spine2);
  line(j.spine2,j.chest);  line(j.chest,j.neck); line(j.neck,j.head);
  line(j.hipL,j.hipR,1.15); line(j.shL,j.shR,1.15);
  line(j.chest,j.shL); line(j.shL,j.elL); line(j.elL,j.haL);
  line(j.chest,j.shR); line(j.shR,j.elR); line(j.elR,j.haR);
  line(j.pelvis,j.hipL); line(j.hipL,j.knL); line(j.knL,j.ftL); line(j.ftL,j.toL);
  line(j.pelvis,j.hipR); line(j.hipR,j.knR); line(j.knR,j.ftR); line(j.ftR,j.toR);

  // ребра
  const m = j.chest.mat;
  const rx = m[0], rz = m[6];
  for (const base of [j.spine2.pos, j.chest.pos]){
    for (let s=-1; s<=1; s+=2){
      items.push({ type:'line', a:base,
        b:[base[0]+rx*0.15*s, base[1]+0.015, base[2]+rz*0.15*s],
        color:col, w:0.8 });
    }
  }
  items.push({ type:'circle', p:j.head.pos, r:0.135, color:col });
}

function drawItems(items){
  const out = [];
  for (const it of items){
    if (it.type === 'line'){
      const A = project(it.a), B = project(it.b);
      if (!A || !B) continue;
      out.push({ z:(A.z+B.z)*0.5, A, B, color:it.color, w:it.w });
    } else {
      const C = project(it.p);
      if (!C) continue;
      out.push({ z:C.z, C, r:it.r*C.s, color:it.color, circle:true });
    }
  }
  out.sort((p,q)=>q.z-p.z);
  ctx.lineCap='round'; ctx.lineJoin='round';

  for (const o of out){
    if (o.circle){
      ctx.beginPath(); ctx.arc(o.C.x,o.C.y,o.r,0,Math.PI*2);
      ctx.fillStyle='#0b0b12'; ctx.fill();
      ctx.globalAlpha=0.22; ctx.lineWidth=7; ctx.strokeStyle=o.color; ctx.stroke();
      ctx.globalAlpha=1; ctx.lineWidth=2.2; ctx.strokeStyle='#eee6d2'; ctx.stroke();
      continue;
    }
    const lw = Math.max(0.9, Math.min(4.5, o.w * o.A.s * 0.022));
    ctx.globalAlpha=0.16; ctx.lineWidth=lw*4.2; ctx.strokeStyle=o.color;
    ctx.beginPath(); ctx.moveTo(o.A.x,o.A.y); ctx.lineTo(o.B.x,o.B.y); ctx.stroke();
    ctx.globalAlpha=1; ctx.lineWidth=lw*1.7; ctx.strokeStyle='#e9e2cf';
    ctx.beginPath(); ctx.moveTo(o.A.x,o.A.y); ctx.lineTo(o.B.x,o.B.y); ctx.stroke();
  }
  ctx.globalAlpha=1;
}

let phase = 'fight';
let phaseTimer = 0;

function resetRound(){
  f1.x=-2.0; f2.x=2.0;
  f1.hp=f1.maxHp; f2.hp=f2.maxHp;
  f1.deathK=0; f2.deathK=0;
  f1.yOff=0; f2.yOff=0; f1.spinYaw=0; f2.spinYaw=0;
  f1.tilt=0; f2.tilt=0; f1.pitch=0; f2.pitch=0;
  f1.knockT=0; f2.knockT=0;
  f1.faceDir= 1; f1.yaw= Math.PI/2;
  f2.faceDir=-1; f2.yaw=-Math.PI/2;
  f1.setState('idle'); f1.idleWait=0.2;
  f2.setState('idle'); f2.idleWait=0.2;
  roundNum++;
  updateHUD();
  banner('РАУНД ' + roundNum, 700);
}

function step(dt){
  if (phase === 'fight'){
    f1.update(dt, f2);
    f2.update(dt, f1);

    const d = f2.x - f1.x;
    if (Math.abs(d) < 0.88){
      const push = (0.88 - Math.abs(d)) * 0.5 * (d>=0?1:-1);
      f1.x -= push; f2.x += push;
    }

    if (f1.hp <= 0 || f2.hp <= 0){
      phase = 'over'; phaseTimer = 0;
      if (f1.hp<=0 && f2.hp<=0){
        banner('НІЧИЯ', 1200);
      } else if (f1.hp<=0){
        scores[1]++; updateHUD();
        banner("ПОЛУМ'Я ПЕРЕМАГАЄ", 1200);
      } else {
        scores[0]++; updateHUD();
        banner('КРИГА ПЕРЕМАГАЄ', 1200);
      }
    }
  } else {
    phaseTimer += dt;
    f1.update(dt, f2);
    f2.update(dt, f1);
    if (phaseTimer > 1.5){
      phase = 'fight';
      resetRound();
    }
  }

  updateSparks(dt);
  updateHUD();

  if (hitFlash > 0) hitFlash = Math.max(0, hitFlash - dt*3.5);
  if (logTimer > 0){ logTimer -= dt; if (logTimer<=0) logEl.textContent = ''; }

  const midX = (f1.x + f2.x) * 0.5;
  cam.pos[0] += (midX*0.32 - cam.pos[0]) * Math.min(1, dt*3.2);
  cam.target[0] = cam.pos[0];

  if (shake > 0){
    shake = Math.max(0, shake - dt*1.8);
    cam.pos[0] += (Math.random()-0.5)*shake;
    cam.pos[1] = 2.25 + (Math.random()-0.5)*shake*1.4;
  } else {
    cam.pos[1] += (2.25 - cam.pos[1]) * Math.min(1, dt*5);
  }
  cam.pos[2] = 6.1 + Math.sin(performance.now()*0.0004)*0.25;

  if (hitFlash > 0) shake = Math.max(shake, hitFlash*0.09);
}

function render(){
  ctx.fillStyle='#07070d';
  ctx.fillRect(0,0,W,H);

  updateBasis();
  drawFloor();
  drawShadow(f1); drawShadow(f2);

  const items = [];
  updateSkeleton(f1); addFighterItems(f1, items);
  updateSkeleton(f2); addFighterItems(f2, items);
  drawItems(items);
  drawSparks();

  // спалах екрану при ударі
  if (hitFlash > 0.01){
    ctx.fillStyle = `rgba(255,90,40,${hitFlash*0.28})`;
    ctx.fillRect(0,0,W,H);
  }

  // віньєтка
  const g = ctx.createRadialGradient(CX,CY,Math.min(W,H)*0.25, CX,CY,Math.max(W,H)*0.75);
  g.addColorStop(0,'rgba(0,0,0,0)');
  g.addColorStop(1,'rgba(0,0,0,0.62)');
  ctx.fillStyle = g;
  ctx.fillRect(0,0,W,H);
}

let lastT = performance.now();
function loop(now){
  requestAnimationFrame(loop);
  let dt = (now-lastT)/1000;
  lastT = now;
  if (dt > 0.05) dt = 0.05;
  step(dt);
  render();
}

/* =====================================================

updateHUD();
banner('БІЙ!', 800);
requestAnimationFrame(loop);
